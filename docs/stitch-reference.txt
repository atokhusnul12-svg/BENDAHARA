<script>
  const SUBSIDI_AWAL = 5700000;
  const TOTAL_SEBELUMNYA = 243000;

  function parseRupiah(str) {
    if (!str) return 0;
    const clean = str.toString().replace(/[^0-9]/g, '');
    return parseInt(clean, 10) || 0;
  }

  function formatRupiah(num) {
    return 'Rp ' + (num || 0).toLocaleString('id-ID');
  }

  function handleHargaInput(input) {
    const rawVal = parseRupiah(input.value);
    input.value = rawVal > 0 ? rawVal.toLocaleString('id-ID') : '';
    recalculateTotal();
  }

  function adjustQty(delta) {
    const qtyInput = document.getElementById('qtyInput');
    let current = parseInt(qtyInput.value, 10) || 1;
    current = Math.max(1, current + delta);
    qtyInput.value = current;
    recalculateTotal();
  }

  function recalculateTotal() {
    const qty = parseInt(document.getElementById('qtyInput').value, 10) || 0;
    const satuan = document.getElementById('satuanSelect').value;
    const hargaSatuan = parseRupiah(document.getElementById('hargaInput').value);
    
    const total = qty * hargaSatuan;
    const sisa = SUBSIDI_AWAL - TOTAL_SEBELUMNYA - total;

    // Update Banner Total
    document.getElementById('totalDisplay').textContent = formatRupiah(total);
    document.getElementById('calcHelper').textContent = 
      `Dihitung otomatis: ${qty} ${satuan} × ${formatRupiah(hargaSatuan)}`;

    // Update Live Preview Box
    document.getElementById('sisaDisplay').textContent = 
      `Sisa subsidi setelah pengeluaran ini: ${formatRupiah(sisa)}`;
    
    document.getElementById('breakdownDisplay').innerHTML = 
      `Subsidi Awal: ${formatRupiah(SUBSIDI_AWAL)} • Total Sebelumnya: ${formatRupiah(TOTAL_SEBELUMNYA)} • Pengeluaran Baru: <span class="text-error font-semibold">-${formatRupiah(total)}</span>`;
  }

  function showToast() {
    const toast = document.getElementById('successToast');
    toast.classList.remove('opacity-0', '-translate-y-4', 'pointer-events-none');
    toast.classList.add('opacity-100', 'translate-y-0');
    setTimeout(() => {
      hideToast();
    }, 3500);
  }

  function hideToast() {
    const toast = document.getElementById('successToast');
    toast.classList.add('opacity-0', '-translate-y-4', 'pointer-events-none');
    toast.classList.remove('opacity-100', 'translate-y-0');
  }

  function handleFormSubmit(event) {
    event.preventDefault();
    showToast();
  }

  function resetOrCancel() {
    if (confirm('Batalkan pengisian form ini?')) {
      document.getElementById('kategoriSelect').value = 'LBM';
      document.getElementById('jenisAcaraInput').value = 'kegiatan lbm';
      document.getElementById('rincianInput').value = 'rokok surya 16';
      document.getElementById('qtyInput').value = '4';
      document.getElementById('satuanSelect').value = 'bungkus';
      document.getElementById('hargaInput').value = '32.000';
      recalculateTotal();
    }
  }

  // Initial calculation check
  recalculateTotal();
</script></main></body></html>

    function dismissToast() {
      const toast = document.getElementById('toastNotification');
      if (toast) {
        toast.classList.add('-translate-y-12', 'opacity-0');
        setTimeout(() => toast.style.display = 'none', 500);
      }
    }

    function triggerToast(message) {
      const toast = document.getElementById('toastNotification');
      if (toast) {
        toast.style.display = 'flex';
        toast.querySelector('.text-body-sm').textContent = message;
        setTimeout(() => {
          toast.classList.remove('-translate-y-12', 'opacity-0');
        }, 10);
        setTimeout(dismissToast, 4000);
      }
    }

    function openCategoryModal(title, defaultVal) {
      document.getElementById('modalTitle').textContent = title;
      document.getElementById('categoryInputName').value = defaultVal || '';
      const modal = document.getElementById('categoryModal');
      modal.classList.remove('hidden');
      modal.classList.add('flex');
    }

    function closeCategoryModal() {
      const modal = document.getElementById('categoryModal');
      modal.classList.add('hidden');
      modal.classList.remove('flex');
    }

    function saveCategory() {
      const val = document.getElementById('categoryInputName').value;
      closeCategoryModal();
      triggerToast(val ? 'Kategori "' + val + '" berhasil diperbarui.' : 'Kategori baru berhasil ditambahkan.');
    }

    function toggleActiveStatus(btn) {
      const circle = btn.querySelector('span');
      if (btn.classList.contains('bg-primary')) {
        btn.classList.remove('bg-primary');
        btn.classList.add('bg-outline-variant');
        circle.classList.remove('translate-x-5');
        circle.classList.add('translate-x-0');
      } else {
        btn.classList.add('bg-primary');
        btn.classList.remove('bg-outline-variant');
        circle.classList.add('translate-x-5');
        circle.classList.remove('translate-x-0');
      }
    }

    function openModal(id) {
      const el = document.getElementById(id);
      if (el) {
        el.classList.remove('hidden');
        el.classList.add('flex');
      }
    }

    function closeModal(id) {
      const el = document.getElementById(id);
      if (el) {
        el.classList.add('hidden');
        el.classList.remove('flex');
      }
    }

    function triggerSyncSaved() {
      closeModal('modalSync');
      triggerToast('Tautan Google Sheets berhasil diperbarui dan disinkronkan.');
    }

    function showBackupPrompt() {
      triggerToast('File backup kas MMD 1448H berhasil diunduh ke perangkat.');
    }

    function showHelpModal() {
      triggerToast('Membuka dokumen panduan operasional bendahara MMD.');
    }

    // Auto dismiss initial toast after 5s
    setTimeout(dismissToast, 4500);
  </script>
